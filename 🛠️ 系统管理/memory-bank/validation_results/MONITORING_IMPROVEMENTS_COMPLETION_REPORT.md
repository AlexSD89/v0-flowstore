```python
# 核心算法
def _calculate_cosine_similarity(self, vec1, vec2):
    common_terms = set(vec1.keys()) & set(vec2.keys())
    if not common_terms:
        return 0
    dot_product = sum(vec1[term] * vec2[term] for term in common_terms)
    return dot_product  # 向量已归一化
```

#### Agent协作流程
```typescript
class GateHRAgentCollaboration {
  private agents: Map<string, HRAgent> = new Map();

  async processHRRequest(request: HRRequest): Promise<HRResponse> {
    // 1. 需求分析和分类
    const classificationAgent = this.agents.get('classification');
    const classification = await classificationAgent.classifyRequest(request);

    // 2. 智能路由到合适的专业Agent
    const routingAgent = this.agents.get('routing');
    const agentAssignment = await routingAgent.assignToAgent(request, classification);

    // 3. 专业Agent处理
    const professionalAgent = this.agents.get(agentAssignment.agentType);
    const processing = await professionalAgent.processRequest(request);

    // 4. 质量检查和优化
    const qualityAgent = this.agents.get('quality');
    const qualityCheck = await qualityAgent.validateQuality(processing);

    return qualityCheck.approved ? processing : await this.refineResponse(processing, qualityCheck.issues);
  }
}
```

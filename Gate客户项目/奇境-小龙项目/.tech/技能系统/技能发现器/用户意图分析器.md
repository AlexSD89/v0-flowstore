```python
class UserIntentAnalyzer:
    def __init__(self):
        self.text_processor = TextProcessor()
        self.intent_classifier = IntentClassifier()
        self.entity_extractor = EntityExtractor()
        self.skill_matcher = SkillMatcher()
        self.personalizer = PersonalizationEngine()
        self.confidence_calculator = ConfidenceCalculator()

    def analyze_user_intent(self, user_input, user_context=None):
        """
        分析用户意图并返回技能推荐
        """
        # 文本预处理
        processed_text = self.text_processor.process(user_input)

        # 意图分类
        intent_classification = self.intent_classifier.classify(processed_text)

        # 实体提取
        entities = self.entity_extractor.extract(processed_text, intent_classification)

        # 技能匹配
        skill_matches = self.skill_matcher.match_skills(
            intent_classification, entities, user_context
        )

        # 个性化优化
        if user_context:
            skill_matches = self.personalizer.personalize_matches(
                skill_matches, user_context
            )

        # 置信度计算
        matches_with_confidence = []
        for match in skill_matches:
            confidence = self.confidence_calculator.calculate(
                match, intent_classification, entities, user_context
            )
            match["confidence"] = confidence
            matches_with_confidence.append(match)

        # 排序和过滤
        final_matches = self._rank_and_filter_matches(matches_with_confidence)

        return {
            "user_intent": {
                "classification": intent_classification,
                "entities": entities,
                "processed_text": processed_text
            },
            "skill_recommendations": final_matches,
            "analysis_metadata": {
                "processing_time": self._get_processing_time(),
                "confidence_distribution": self._get_confidence_distribution(final_matches),
                "match_quality_score": self._calculate_match_quality(final_matches)
            }
        }

    def _rank_and_filter_matches(self, matches_with_confidence):
        """排序和过滤匹配结果"""
        # 过滤低置信度匹配
        filtered_matches = [
            match for match in matches_with_confidence
            if match["confidence"] >= 0.6
        ]

        # 按置信度和相关性排序
        ranked_matches = sorted(
            filtered_matches,
            key=lambda x: (x["confidence"], x["relevance_score"]),
            reverse=True
        )

        # 保证多样性
        diversified_matches = self._ensure_diversity(ranked_matches)

        return diversified_matches[:10]  # 返回前10个最佳匹配

    def _ensure_diversity(self, matches):
        """确保推荐结果多样性"""
        if not matches:
            return matches

        diversified = []
        used_categories = set()

        # 优先选择不同类别的技能
        for match in matches:
            category = match.get("category", "other")
            if category not in used_categories or len(diversified) < 5:
                diversified.append(match)
                used_categories.add(category)

        # 补充其他高相关技能
        for match in matches:
            if match not in diversified and len(diversified) < 10:
                diversified.append(match)

        return diversified
```

### 智能文本处理器
```python
class TextProcessor:
    def __init__(self):
        self.cleaner = TextCleaner()
        self.tokenizer = AdvancedTokenizer()
        self.embedder = SemanticEmbedder()
        self.normalizer = TextNormalizer()

    def process(self, user_input):
        """处理用户输入文本"""
        # 文本清理
        cleaned_text = self.cleaner.clean(user_input)

        # 文本标准化
        normalized_text = self.normalizer.normalize(cleaned_text)

        # 分词和词性标注
        tokens = self.tokenizer.tokenize(normalized_text)

        # 语义向量计算
        semantic_vector = self.embedder.embed(normalized_text)

        # 关键词提取
        keywords = self._extract_keywords(tokens, semantic_vector)

        return {
            "original_text": user_input,
            "cleaned_text": cleaned_text,
            "normalized_text": normalized_text,
            "tokens": tokens,
            "semantic_vector": semantic_vector,
            "keywords": keywords,
            "language": self._detect_language(normalized_text),
            "text_type": self._detect_text_type(normalized_text)
        }

    def _extract_keywords(self, tokens, semantic_vector):
        """提取关键词"""
        keywords = []

        # 基于TF-IDF的关键词
        tfidf_keywords = self._tfidf_extract(tokens)
        keywords.extend(tfidf_keywords)

        # 基于语义向量的关键词
        semantic_keywords = self._semantic_extract(semantic_vector)
        keywords.extend(semantic_keywords)

        # 去重和排序
        unique_keywords = list(set(keywords))
        ranked_keywords = sorted(
            unique_keywords,
            key=lambda x: self._calculate_keyword_score(x, tokens, semantic_vector),
            reverse=True
        )

        return ranked_keywords[:20]
```

class IntentClassifier:
    def __init__(self):
        self.intent_categories = self._load_intent_categories()
        self.classifier_model = self._load_classifier_model()
        self.rule_engine = RuleEngine()
        self.context_analyzer = ContextAnalyzer()

    def classify(self, processed_text):
        """分类用户意图"""
        # 基于规则的快速分类
        rule_based_intent = self.rule_engine.classify(processed_text)

        # 基于机器学习的分类
        ml_intent = self.classifier_model.predict(processed_text)

        # 上下文增强分类
        context_enhanced_intent = self.context_analyzer.enhance_classification(
            processed_text, rule_based_intent, ml_intent
        )

        # 置信度计算
        classification_confidence = self._calculate_classification_confidence(
            rule_based_intent, ml_intent, context_enhanced_intent
        )

        return {
            "primary_intent": context_enhanced_intent["primary"],
            "secondary_intents": context_enhanced_intent["secondary"],
            "intent_hierarchy": context_enhanced_intent["hierarchy"],

class EntityExtractor:
    def __init__(self):
        self.ner_model = self._load_ner_model()
        self.pattern_matcher = PatternMatcher()
        self.domain_extractor = DomainExtractor()
        self.parameter_parser = ParameterParser()

    def extract(self, processed_text, intent_classification):
        """提取实体信息"""
        entities = {}

        # 通用实体提取
        general_entities = self.ner_model.extract(processed_text["normalized_text"])
        entities["general"] = general_entities

        # 意图特定实体提取
        intent_specific_entities = self._extract_intent_specific_entities(
            processed_text, intent_classification
        )
        entities["intent_specific"] = intent_specific_entities

        # 领域实体提取
        domain_entities = self.domain_extractor.extract(
            processed_text, intent_classification
        )
        entities["domain"] = domain_entities

        # 参数解析
        parameters = self.parameter_parser.parse(
            processed_text, intent_classification, entities
        )
        entities["parameters"] = parameters

        # 实体验证和清洗
        validated_entities = self._validate_and_clean_entities(entities)

        return validated_entities

    def _extract_intent_specific_entities(self, processed_text, intent_classification):
        """提取意图特定实体"""
        primary_intent = intent_classification["primary_intent"]

        if primary_intent == "data_analysis":
            return self._extract_data_analysis_entities(processed_text)
        elif primary_intent == "design_review":
            return self._extract_design_review_entities(processed_text)
        elif primary_intent == "project_management":
            return self._extract_project_management_entities(processed_text)
        elif primary_intent == "content_creation":
            return self._extract_content_creation_entities(processed_text)
        elif primary_intent == "automation":
            return self._extract_automation_entities(processed_text)
        else:
            return {}

    def _extract_data_analysis_entities(self, processed_text):
        """提取数据分析特定实体"""
        entities = {
            "data_sources": [],
            "analysis_types": [],
            "output_formats": [],
            "quality_requirements": []
        }

        # 数据源识别
        data_source_patterns = [
            r"(?:Excel|表格|数据文件)\s*[:：]?\s*([^\s,，。.]+)",
            r"文件[：:]?\s*([^\s,，。.]+\.(?:xlsx|xls|csv))",
            r"数据来源[：:]?\s*([^\s,，。.]+)"
        ]

        for pattern in data_source_patterns:
            matches = re.findall(pattern, processed_text["normalized_text"], re.IGNORECASE)
            entities["data_sources"].extend(matches)

        # 分析类型识别
        analysis_patterns = [
            r"(?:统计|分析|计算)\s*([^\s,，。.]+)",
            r"生成\s*(?:图表|报表|统计)\s*(?:关于|的)\s*([^\s,，。.]+)",
            r"比较\s*([^\s,，。.]+)"
        ]

        for pattern in analysis_patterns:
            matches = re.findall(pattern, processed_text["normalized_text"], re.IGNORECASE)
            entities["analysis_types"].extend(matches)

        # 输出格式识别
        format_patterns = [
            r"(?:输出|生成|创建)\s*(?:图表|表格|报告|PDF|Word|Excel)",
            r"(?:格式|形式)[：:]?\s*([^\s,，。.]+)"
        ]

        for pattern in format_patterns:
            if re.search(pattern, processed_text["normalized_text"], re.IGNORECASE):
                entities["output_formats"].append("chart")  # 默认图表格式

        return entities
```

class IntentAnalysisCache:
    def __init__(self):
        self.query_cache = LRUCache(maxsize=1000)

        """获取缓存结果"""
        if user_input_hash in self.query_cache:
            return self.query_cache[user_input_hash]
        return None

    def cache_result(self, user_input_hash, result, ttl=3600):
        """缓存分析结果"""
        self.query_cache[user_input_hash] = result
        # 可以添加Redis缓存用于分布式部署

    def get_similar_queries(self, processed_text):
        """获取相似查询"""
        # 基于语义相似度查找相似查询
        similar_queries = []
        cached_queries = self.query_cache.keys()

        for cached_hash in cached_queries:
            cached_result = self.query_cache[cached_hash]
            similarity = self._calculate_semantic_similarity(
                processed_text, cached_result["processed_text"]
            )
            if similarity > 0.8:
                similar_queries.append({
                    "hash": cached_hash,
                    "similarity": similarity,
                    "result": cached_result
                })

        return sorted(similar_queries, key=lambda x: x["similarity"], reverse=True)
```

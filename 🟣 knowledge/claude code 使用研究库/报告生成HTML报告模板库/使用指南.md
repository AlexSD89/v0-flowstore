## 📋 快速开始

### 1. 基础文件结构

```
your-project/
├── 基础模板.html           # 主模板文件
├── 图表生成器.js           # 图表生成工具
├── 数据模板.js             # 数据结构模板
├── your-data.js           # 你的数据文件
└── output.html            # 最终生成的报告
```

### 2. 最简单的使用方式

```html
<!-- 步骤1: 复制基础模板.html并重命名 -->
<!DOCTYPE html>
<html lang="zh-CN">
<head>
    <!-- 保持原始head内容 -->
</head>
<body>
    <!-- 你的内容将在这里 -->
</body>
</html>

<!-- 步骤2: 在页面底部添加你的数据 -->
<script src="数据模板.js"></script>
<script src="图表生成器.js"></script>
<script src="your-data.js"></script>
</html>
```

```javascript
// data-loader.js
class ReportLoader {
    constructor() {
        this.apiEndpoint = '/api/report-data'; // 你的API端点
    }

    async loadReportData(reportId) {
        try {
            // 从API加载数据
            const response = await fetch(`${this.apiEndpoint}/${reportId}`);
            const data = await response.json();

            // 验证数据
            const validation = this.validateData(data);
            if (!validation.isValid) {
                console.error('数据验证失败:', validation.errors);
                return null;
            }

            // 创建报告
            return this.createReport(data);

        } catch (error) {
            console.error('加载报告数据失败:', error);
            return null;
        }
    }

    validateData(data) {
        // 实现你的数据验证逻辑
        return { isValid: true, errors: [] };
    }

    createReport(data) {
        const report = new MarketAnalysisTemplate();
        report.metadata = data.metadata;
        report.keyMetrics = data.keyMetrics;
        report.executiveSummary = data.executiveSummary;

        // 设置各个图表的数据
        data.sections.forEach(section => {
            switch (section.type) {
                case 'marketSize':
                    report.setMarketSizeData(
                        section.data.categories,
                        section.data.marketSize,
                        section.data.newSupply,
                        section.data.vacancyRate
                    );
                    break;
                case 'tenantStructure':
                    report.setTenantStructureData(section.data);
                    break;
                // 其他章节类型...
            }
        });

        return report;
    }
}
```

```javascript
// custom-template.js
class CustomReportTemplate extends MarketAnalysisTemplate {
    constructor() {
        super();
        this.customSections = [];
    }

    // 添加自定义章节
    addCustomSection(title, content, size = 'medium') {
        this.sections.push({
            title,
            size,
            content: typeof content === 'string' ? content : content()
        });
        return this;
    }

    // 添加自定义图表
    addCustomChart(title, chartId, chartData, chartType = 'mixed') {
        this.sections.push({
            title,
            size: 'large',
            chart: {
                id: chartId,
                type: chartType,
                data: chartData,
                explanation: this.generateChartExplanation(chartData)
            }
        });
        return this;
    }

    // 生成图表说明
    generateChartExplanation(data) {
        // 基于数据智能生成说明文字
        if (data.series && data.series.length > 0) {
            const mainSeries = data.series[0];
            const trend = this.analyzeTrend(mainSeries.data);
            return `数据显示${mainSeries.name}呈现${trend}趋势，${this.generateInsight(data)}`;
        }
        return '数据分析图表，展示关键指标变化趋势。';
    }

    analyzeTrend(data) {
        if (data.length < 2) return '稳定';
        const first = data[0];
        const last = data[data.length - 1];
        const change = ((last - first) / first * 100).toFixed(1);

        if (change > 10) return '快速增长';
        if (change > 0) return '稳步增长';
        if (change > -10) return '小幅下降';
        return '明显下降';
    }

    generateInsight(data) {
        // 基于数据特征生成洞察
        return '建议关注数据波动性和季节性变化规律。';
    }
}
```

### 2. 响应式图表处理

```javascript
class ResponsiveChartManager extends ReportChartGenerator {
    constructor() {
        super();
        this.breakpoints = {
            mobile: 480,
            tablet: 768,
            desktop: 1024
        };
        this.currentBreakpoint = this.getCurrentBreakpoint();
        this.setupResponsiveHandling();
    }

    getCurrentBreakpoint() {
        const width = window.innerWidth;
        if (width <= this.breakpoints.mobile) return 'mobile';
        if (width <= this.breakpoints.tablet) return 'tablet';
        if (width <= this.breakpoints.desktop) return 'desktop';
        return 'large';
    }

    setupResponsiveHandling() {
        window.addEventListener('resize', _.debounce(() => {
            const newBreakpoint = this.getCurrentBreakpoint();
            if (newBreakpoint !== this.currentBreakpoint) {
                this.currentBreakpoint = newBreakpoint;
                this.handleBreakpointChange();
            }
            this.resize();
        }, 300));
    }

    handleBreakpointChange() {
        // 根据断点调整图表配置
        this.charts.forEach((chart, containerId) => {
            const option = chart.getOption();

            if (this.currentBreakpoint === 'mobile') {
                // 移动端调整
                option.legend[0].top = '5%';
                option.grid[0].top = '20%';
                option.xAxis[0].axisLabel.rotate = 45;
            } else {
                // 桌面端恢复
                option.legend[0].top = '10%';
                option.grid[0].top = '15%';
                option.xAxis[0].axisLabel.rotate = 0;
            }

            chart.setOption(option, false);
        });
    }
}
```

### 3. 数据动态更新

```javascript
class DynamicReportUpdater {
    constructor(chartManager) {
        this.chartManager = chartManager;
        this.updateInterval = null;
    }

    // 设置定时更新
    startAutoUpdate(intervalMs = 30000) {
        this.updateInterval = setInterval(() => {
            this.updateData();
        }, intervalMs);
    }

    // 停止自动更新
    stopAutoUpdate() {
        if (this.updateInterval) {
            clearInterval(this.updateInterval);
            this.updateInterval = null;
        }
    }

    // 更新数据
    async updateData() {
        try {
            const newData = await this.fetchLatestData();
            this.updateCharts(newData);
            this.updateMetrics(newData);
        } catch (error) {
            console.error('数据更新失败:', error);
        }
    }

    async fetchLatestData() {
        // 实现数据获取逻辑
        const response = await fetch('/api/latest-data');
        return response.json();
    }

    updateCharts(newData) {
        Object.keys(newData.charts).forEach(chartId => {
            const chart = this.chartManager.getChart(chartId);
            if (chart) {
                const option = chart.getOption();
                // 更新数据
                newData.charts[chartId].series.forEach((series, index) => {
                    option.series[index].data = series.data;
                });
                chart.setOption(option, false);
            }
        });
    }

    updateMetrics(newData) {
        // 更新关键指标
        Object.keys(newData.metrics).forEach(metricId => {
            const element = document.getElementById(metricId);
            if (element) {
                element.textContent = newData.metrics[metricId];
            }
        });
    }
}
```

### 1. 图表不显示

**问题**: 图表容器存在但图表不渲染

```javascript
// 解决方案：数据验证
function validateChartData(data) {
    if (!Array.isArray(data.series)) {
        console.error('series必须是数组');
        return false;
    }

    return data.series.every(series =>
        Array.isArray(series.data) &&
        series.data.every(value => typeof value === 'number')
    );
}
```
